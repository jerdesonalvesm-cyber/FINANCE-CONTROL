drop policy if exists "Cada usuario ve suas transacoes" on public.transacoes; drop policy if exists "Cada usuario cadastra suas transacoes" on public.transacoes; drop policy if exists "Cada usuario altera suas transacoes" on public.transacoes; drop policy if exists "Cada usuario exclui suas transacoes" on public.transacoes; alter table public.transacoes enable row level security; create policy "Cada usuario ve suas transacoes" on public.transacoes for select to authenticated using (auth.uid() = user_id); create policy "Cada usuario cadastra suas transacoes" on public.transacoes for insert to authenticated with check (auth.uid() = user_id); create policy "Cada usuario altera suas transacoes" on public.transacoes for update to authenticated using (auth.uid() = user_id) with check (auth.uid() = user_id); create policy "Cada usuario exclui suas transacoes" on public.transacoes for delete to authenticated using (auth.uid() = user_id); grant select, insert, update, delete on public.transacoes to authenticated;
